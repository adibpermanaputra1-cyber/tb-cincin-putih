// Generate complete ESC/POS receipt byte data export function generateEscPosReceipt( transaction: SaleTransaction, settings: StoreSettings, paperWidth: 58 | 80 = 58 ): Uint8Array { const colWidth = paperWidth === 80 ? 48 : 32; const builder = new EscPosBuilder(); const isKasbon = transaction.paymentMethod === 'KASBON'; // Header Toko (Centered) builder.align(1); builder.doubleSize(true); builder.bold(true); builder.textLine(settings.storeName || 'TB. CINCIN PUTIH'); builder.doubleSize(false); builder.bold(false); if (settings.tagline) builder.textWrapped(settings.tagline, colWidth); if (settings.address) builder.textWrapped(settings.address, colWidth); if (settings.phone) builder.textWrapped(`Telp: ${settings.phone}`, colWidth); if (isKasbon) { builder.feed(1); builder.bold(true); builder.textLine('*** NOTA BON KASBON / HUTANG ***'); builder.bold(false); } builder.divider('=', colWidth); // Metadata Transaksi (Left Aligned) builder.align(0); builder.twoColumn('No. Faktur:', transaction.invoiceNo || 'INV-001', colWidth); builder.twoColumn('Tanggal:', formatDateOnly(transaction.date), colWidth); builder.twoColumn('Kasir:', transaction.cashierName || 'Kasir', colWidth); builder.twoColumn('Pelanggan:', transaction.customerName || 'Pelanggan Umum', colWidth); if (transaction.customerPhone) { builder.twoColumn('No. HP:', transaction.customerPhone, colWidth); } builder.divider('-', colWidth); // Items Header builder.bold(true); builder.twoColumn('MATERIAL / ITEM', 'SUBTOTAL', colWidth); builder.bold(false); builder.divider('-', colWidth); // ======================================================================= // 🛒 PERBAIKAN LOOPING ITEM: Fallback multi-alias properti objek item // ======================================================================= const items = (transaction.items || (transaction as any).cart || []) as any[]; items.forEach((item) => { // 1. Ekstraksi Nama Barang dengan Fallback const rawName = item.productName || item.name || item.nama_barang || item.nama || item.title || 'Barang Material'; // 2. Ekstraksi Qty / Jumlah const rawQty = Number(item.quantity ?? item.qty ?? item.jumlah ?? 1); // 3. Ekstraksi Satuan Unit const rawUnit = item.unit || item.sellingUnit || item.selectedUnit || item.satuan || 'Pcs'; // 4. Ekstraksi Harga Satuan const rawPrice = Number( item.unitPrice ?? item.activePrice ?? item.price ?? item.harga ?? item.sellPrice ?? 0 ); // 5. Ekstraksi Subtotal const rawSubtotal = Number(item.subtotal ?? (rawQty * rawPrice)); // Baris 1: Nama Material (Bold) builder.bold(true); builder.textLine(rawName); builder.bold(false); // Baris 2: Qty Satuan x Harga Satuan (Kiri) & Subtotal (Kanan) const qtyPrice = ` ${rawQty} ${rawUnit} x ${formatRupiah(rawPrice)}`; const subtotalStr = formatRupiah(rawSubtotal); builder.twoColumn(qtyPrice, subtotalStr, colWidth); }); builder.divider('-', colWidth); // Subtotal & Total Pembayaran builder.twoColumn('Subtotal:', formatRupiah(transaction.subtotal), colWidth); if (transaction.discountAmount > 0) { builder.twoColumn('Diskon:', `-${formatRupiah(transaction.discountAmount)}`, colWidth); } if (transaction.deliveryFee > 0) { builder.twoColumn('Ongkir:', formatRupiah(transaction.deliveryFee), colWidth); } builder.bold(true); builder.divider('=', colWidth); builder.twoColumn('TOTAL:', formatRupiah(transaction.grandTotal), colWidth); builder.bold(false); builder.divider('-', colWidth); // Informasi Pembayaran / Kasbon if (isKasbon) { builder.twoColumn('Metode:', 'KASBON / TEMPO', colWidth); builder.twoColumn('DP Masuk:', formatRupiah(transaction.paidAmount), colWidth); builder.bold(true); builder.twoColumn('SISA KASBON:', formatRupiah(transaction.remainingAmount), colWidth); builder.bold(false); if (transaction.dueDate) { builder.twoColumn('Jatuh Tempo:', formatDateOnly(transaction.dueDate), colWidth); } builder.feed(1); builder.align(1); builder.textLine('Ttd Penerima Ttd Kasir'); builder.feed(2); builder.textLine('( ............. ) ( ............. )'); } else { builder.twoColumn('Metode Bayar:', transaction.paymentMethod, colWidth); builder.twoColumn('Tunai Diterima:', formatRupiah(transaction.cashPaid), colWidth); builder.bold(true); builder.twoColumn('Kembalian:', formatRupiah(transaction.changeDue), colWidth); builder.bold(false); } builder.divider('-', colWidth); // Footer Struk builder.align(1); const footerText = settings.receiptFooter || settings.footerNote; if (footerText) { builder.textWrapped(footerText, colWidth); } builder.textLine('*** TERIMA KASIH ***'); builder.cut(); return builder.getBytes(); }
